# Runtime dependencies for the deployed Streamlit app (app.py).
#
# Streamlit Community Cloud auto-installs from this file. Keep it lean —
# document-parsing / ingestion deps live in requirements-ingest.txt and
# are installed only in developer environments that run scripts/ingest.py.
#
# NumPy is pinned to 1.26.x to stay compatible with the sentence-transformers
# 4.1 / torch 2.x wheels currently on PyPI. Upgrading past 2.x would require
# rebuilding every downstream wheel.

# Chat UI
streamlit==1.39.0

# LLM + embeddings clients used by app.py
openai==1.55.3
langchain==0.3.7
langchain-community==0.3.5
langchain-openai==0.2.8
langchain-qdrant==0.2.0
langchain-text-splitters==0.3.2

# Cross-encoder reranker used by app.py::rerank_docs. Downloads a ~90 MB
# model on first request. CPU-only inference — no GPU needed.
sentence-transformers==4.1.0

# Vector store client
qdrant-client==1.12.1

# Numerical dep pin.
numpy==1.26.4

# Pillow pin: sentence-transformers>=4.x requires pillow>=10. Pillow 10.x
# has no Python 3.13+ wheels and fails to compile from source on Streamlit
# Cloud. Pillow 11.x ships pre-built wheels for Python 3.13/3.14 so we
# pin here to avoid a source-build failure if the resolver picks 10.x.
pillow>=11.0.0

# Protobuf pin: qdrant-client 1.12.x needs 5.x; the default resolver
# sometimes picks 3.x and produces a runtime import error.
protobuf==5.29.5

# Local .env loader — no-op on Streamlit Cloud but useful for local dev.
python-dotenv==1.0.1
